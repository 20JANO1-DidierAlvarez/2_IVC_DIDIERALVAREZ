// Ejercicio 17 - Contar billetes de $100
using System;

class Program
{
    static void Main()
    {
        int billete, cantidad = 0, total = 0;
        char seguir;

        Console.WriteLine("Ingrese denominaciones ($20, $50, $100). Termine con N");

        do
        {
            Console.Write("Valor del billete: ");
            billete = int.Parse(Console.ReadLine());

            if (billete == 100)
            {
                cantidad++;
                total = total + 100;
            }

            Console.Write("¿Ingresar otro? (S/N): ");
            seguir = Console.ReadKey().KeyChar;
            Console.WriteLine(""); // Salto de línea

        } while (seguir == 'S' || seguir == 's');

        Console.WriteLine("Cantidad de billetes de $100: " + cantidad);
        Console.WriteLine("Total acumulado: $" + total);
    }
}
