// Ejercicio 2 - Primeros 20 múltiplos
using System;

class Program
{
    static void Main()
    {
        int numero;

        Console.Write("Ingrese un número entero: ");
        numero = int.Parse(Console.ReadLine());

        Console.WriteLine("Los primeros 20 múltiplos de " + numero + " son:");

        // Repetimos exactamente 20 veces
        for (int i = 1; i <= 20; i++)
        {
            int resultado = numero * i;
            Console.WriteLine(resultado);
        }
    }
}
