// Ejercicio 19 - Del 300 al 1: primos y suma pares
using System;

class Program
{
    static void Main()
    {
        int contadorPrimos = 0, sumaPares = 0;

        // Recorremos desde 300 hasta 1
        for (int numero = 300; numero >= 1; numero--)
        {
            // Sumar pares
            if (numero % 2 == 0)
            {
                sumaPares = sumaPares + numero;
            }

            // Verificar si es primo
            bool esPrimo = true;
            if (numero > 1)
            {
                for (int i = 2; i < numero; i++)
                {
                    if (numero % i == 0)
                    {
                        esPrimo = false;
                        break;
                    }
                }

                if (esPrimo)
                {
                    contadorPrimos++;
                }
            }
        }

        Console.WriteLine("Cantidad de números primos: " + contadorPrimos);
        Console.WriteLine("Suma total de números pares: " + sumaPares);
    }
}
