// Ejercicio 3 - Sumar primos del 1 al 50
using System;

class Program
{
    static void Main()
    {
        int acumulador = 0; // Variable acumuladora

        // Recorremos del 2 al 50 (el 1 no es primo)
        for (int numero = 2; numero <= 50; numero++)
        {
            bool esPrimo = true; // Suponemos que es primo

            // Revisamos si tiene algún divisor además de 1 y él mismo
            for (int divisor = 2; divisor < numero; divisor++)
            {
                if (numero % divisor == 0)
                {
                    esPrimo = false; // No es primo
                    break; // Salimos del ciclo
                }
            }

            // Si SÍ es primo, lo sumamos al acumulador
            if (esPrimo)
            {
                acumulador = acumulador + numero;
            }
        }

        Console.WriteLine("La suma de todos los números primos entre 1 y 50 es: " + acumulador);
    }
}
