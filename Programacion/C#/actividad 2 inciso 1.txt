using System;

class Program
{
    static void Main()
    {
        int d;

        Console.WriteLine("Ingrese un número del 1 al 7:");
        d = Convert.ToInt32(Console.ReadLine());

        if (d == 6 || d == 7)
            Console.WriteLine("Fin de semana");
        else if (d >= 1 && d <= 5)
            Console.WriteLine("Entre semana");
        else
            Console.WriteLine("Número inválido");
    }
}
