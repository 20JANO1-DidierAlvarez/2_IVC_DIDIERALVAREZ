int op;
Console.WriteLine("1 Ventas\n2 Soporte\n3 Facturacion\n4 Otros");
op = Convert.ToInt32(Console.ReadLine());

if (op == 1)
    Console.WriteLine("Ventas");
else if (op == 2)
    Console.WriteLine("Soporte");
else if (op == 3)
    Console.WriteLine("Facturacion");
else if (op == 4)
    Console.WriteLine("Otros");
else
    Console.WriteLine("Opcion invalida");
