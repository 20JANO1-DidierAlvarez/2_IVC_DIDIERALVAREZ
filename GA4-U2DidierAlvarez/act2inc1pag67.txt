// Ejercicio 18 - Múltiplos de 7, suma y pares
using System;

class Program
{
    static void Main()
    {
        int suma = 0, pares = 0;

        for (int i = 1; i <= 20; i++)
        {
            int multiplo = 7 * i;
            suma = suma + multiplo;

            if (multiplo % 2 == 0)
            {
                pares++;
            }
        }

        Console.WriteLine("Suma total de los múltiplos: " + suma);
        Console.WriteLine("Cantidad de números pares: " + pares);
    }
}
